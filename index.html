# My Blender App - GitHub Version v1.0 import pygame import math pygame.init() screen = pygame.display.set_mode((900,600)) pygame.display.set_caption("MY BLENDER APP - GitHub Build") clock = pygame.time.Clock() objects = [] # list of cubes selected = None angle_x, angle_y = 0.5, 0.5 mode = "OBJECT" # OBJECT, EDIT, SAVE class Cube: def __init__(self, x,y,z): self.x, self.y, self.z = x,y,z self.size = 50 def rot(point, ax, ay): x,y,z = point # X rot y2 = y*math.cos(ax)-z*math.sin(ax) z2 = y*math.sin(ax)+z*math.cos(ax) # Y rot x3 = x2 if 'x2' not in locals() else x*math.cos(ay)-z2*math.sin(ay) # Simplified return (x*math.cos(ay)-z2*math.sin(ay), y2, x*math.sin(ay)+z2*math.cos(ay)) def project(p): x,y,z = p f = 400/(400+z) return (int(x*f+450), int(y*f+300)) def draw_cube(c, color=(0,255,0)): s=c.size/2 corners=[(c.x-s,c.y-s,c.z-s),(c.x+s,c.y-s,c.z-s),(c.x+s,c.y+s,c.z-s),(c.x-s,c.y+s,c.z-s), (c.x-s,c.y-s,c.z+s),(c.x+s,c.y-s,c.z+s),(c.x+s,c.y+s,c.z+s),(c.x-s,c.y+s,c.z+s)] rc=[rot(p, angle_x, angle_y) for p in corners] pc=[project(p) for p in rc] edges=[(0,1),(1,2),(2,3),(3,0),(4,5),(5,6),(6,7),(7,4),(0,4),(1,5),(2,6),(3,7)] for a,b in edges: pygame.draw.line(screen, color, pc[a], pc[b], 2) running=True while running: for e in pygame.event.get(): if e.type==pygame.QUIT: running=False if e.type==pygame.KEYDOWN: if e.key==pygame.K_n: # N = New Cube like Blender objects.append(Cube(0,0,0)) if e.key==pygame.K_s: # S = Save OBJ with open("my_model.obj","w") as f: for obj in objects: s=obj.size/2 f.write(f"v {obj.x-s} {obj.y-s} {obj.z-s}\n") print("SAVED my_model.obj") if e.type==pygame.MOUSEMOTION and pygame.mouse.get_pressed()[0]: angle_y+=e.rel[0]*0.01 angle_x+=e.rel[1]*0.01 screen.fill((30,30,30)) # UI like Blender pygame.draw.rect(screen, (50,50,50), (0,0,150,600)) font=pygame.font.Font(None,22) screen.blit(font.render("MY BLENDER", True, (255,255,255)), (20,20)) screen.blit(font.render("N = New Cube", True, (200,200,200)), (10,60)) screen.blit(font.render("Drag = Rotate", True, (200,200,200)), (10,90)) screen.blit(font.render("S = Save OBJ", True, (200,200,200)), (10,120)) screen.blit(font.render(f"Objects: {len(objects)}", True, (0,255,0)), (10,160)) for obj in objects: draw_cube(obj) if not objects: screen.blit(pygame.font.Font(None,40).render("Press N to Add Cube - Like Blender", True, (255,255,255)), (200,250)) pygame.display.flip() clock.tick(60) pygame.quit()
